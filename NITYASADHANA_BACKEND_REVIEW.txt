@app.get("/api/db-health")
def database_health_check():
    try:
        with engine.connect() as connection:
            database_name = connection.execute(
                text("SELECT DATABASE()")
            ).scalar()

        return {
            "status": "connected",
            "database": database_name,
            "message": "MySQL connection successful"
        }

    except SQLAlchemyError:
        raise HTTPException(
            status_code=503,
            detail="MySQL connection failed. Check database configuration."
        )

@app.get("/api/models-health")
def models_health():
    try:
        tables = inspect(engine).get_table_names()

        expected_tables = [
            "users",
            "devotee_profiles",
            "japa_records",
            "book_reading",
            "hearing_records",
            "seva_records",
            "daily_reports",
        ]

        found_tables = [
            table for table in expected_tables
            if table in tables
        ]

        return {
            "status": "success" if len(found_tables) == 7 else "incomplete",
            "models_registered": 7,
            "tables_found": len(found_tables),
            "tables": found_tables,
            "message": "NityaSadhana database models verification completed"
        }

    except Exception as e:
        raise HTTPException(
            status_code=503,
            detail="Database model verification failed"
        )

===== DATABASE.PY =====
import os
from pathlib import Path

from dotenv import load_dotenv
from sqlalchemy import URL, create_engine
from sqlalchemy.orm import declarative_base, sessionmaker

BASE_DIR = Path(__file__).resolve().parent
load_dotenv(BASE_DIR / ".env")

DB_HOST = os.getenv("DB_HOST")
DB_PORT = int(os.getenv("DB_PORT", "3306"))
DB_NAME = os.getenv("DB_NAME")
DB_USER = os.getenv("DB_USER")
DB_PASSWORD = os.getenv("DB_PASSWORD")

if not all([DB_HOST, DB_NAME, DB_USER, DB_PASSWORD]):
    raise ValueError("Database configuration is incomplete. Check backend/.env")

DATABASE_URL = URL.create(
    drivername="mysql+pymysql",
    username=DB_USER,
    password=DB_PASSWORD,
    host=DB_HOST,
    port=DB_PORT,
    database=DB_NAME,
)

engine = create_engine(
    DATABASE_URL,
    pool_pre_ping=True,
)

SessionLocal = sessionmaker(
    autocommit=False,
    autoflush=False,
    bind=engine,
)

Base = declarative_base()
def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

from sqlalchemy import (
    Column,

    FetchedValue,

from sqlalchemy.dialects.mysql import BIGINT, INTEGER, ENUM

from database import Base


# 1. USERS TABLE
class User(Base):
    __tablename__ = "users"

    user_id = Column(BIGINT(unsigned=True), primary_key=True, autoincrement=True)
    email = Column(String(255), nullable=False, unique=True)
    password_hash = Column(String(255), nullable=False)

    role = Column(
        ENUM("devotee", "admin"),
        server_default=text("'devotee'")
    )

    account_status = Column(
        ENUM("active", "inactive"),
        server_default=text("'active'")
    )

    created_at = Column(
        TIMESTAMP,
        nullable=True,
        server_default=text("CURRENT_TIMESTAMP")
    )

    updated_at = Column(
        TIMESTAMP,
        nullable=True,
        server_default=text("CURRENT_TIMESTAMP"),
        server_onupdate=FetchedValue()
    )


# 2. DEVOTEE PROFILES TABLE
class DevoteeProfile(Base):
    __tablename__ = "devotee_profiles"

    profile_id = Column(BIGINT(unsigned=True), primary_key=True, autoincrement=True)

    user_id = Column(
        BIGINT(unsigned=True),
        ForeignKey("users.user_id", name="fk_profile_user", ondelete="CASCADE"),
        nullable=False,
        unique=True
    )

    full_name = Column(String(150), nullable=False)
    initiated_name = Column(String(150), nullable=True)
    phone = Column(String(20), nullable=True)
    city = Column(String(100), nullable=True)

    created_at = Column(
        TIMESTAMP,
        nullable=True,
        server_default=text("CURRENT_TIMESTAMP")
    )

    updated_at = Column(
        TIMESTAMP,
        nullable=True,
        server_default=text("CURRENT_TIMESTAMP"),
        server_onupdate=FetchedValue()
    )


# 3. JAPA RECORDS TABLE
class JapaRecord(Base):
    __tablename__ = "japa_records"

    __table_args__ = (
        UniqueConstraint(
            "user_id",
            "record_date",
            name="unique_user_japa_date"
        ),
    )

    japa_id = Column(BIGINT(unsigned=True), primary_key=True, autoincrement=True)

    user_id = Column(
        BIGINT(unsigned=True),
        ForeignKey("users.user_id", name="fk_japa_user", ondelete="CASCADE"),
        nullable=False
    )

    record_date = Column(Date, nullable=False)

    rounds_completed = Column(
        INTEGER(unsigned=True),
        nullable=False,
        server_default=text("0")
    )

    notes = Column(Text, nullable=True)

    created_at = Column(
        TIMESTAMP,
        nullable=True,
        server_default=text("CURRENT_TIMESTAMP")
    )
